import HeaderParamFields from '/snippets/en/exode-api/header-params-fields.mdx';

<HeaderParamFields/>

```
GET /saas/v2/form/layout/list
```

Requires authentication and the [**"Forms management"**](/en/exode-api/permissions) permission (`FormManage`).

<Info>
    All filter parameters are optional. Without filters, the method returns all of the school's form layouts, paginated.
    The layout structure is the same as in the [`form/layout/create`](/en/exode-api/school/form-layout/create) response.
</Info>

<Info>
    Array parameters are passed by repeating the parameter: `modes=Form&modes=Signup`.
</Info>

## Request parameters

### Pagination

<ParamField query="take" type="integer" required={false}>
    Number of records per page. From `1` to `1000`. Defaults to `100`.
</ParamField>

<ParamField query="page" type="integer" required={false}>
    Page number (starting at `1`). An alternative to `skip`.
</ParamField>

<ParamField query="skip" type="integer" required={false}>
    Number of records to skip. Defaults to `0`.
</ParamField>

### Filtering

<ParamField query="layoutIds" type="integer[]" required={false}>
    Filter by layout IDs.
</ParamField>

<ParamField query="layoutUuids" type="string[]" required={false}>
    Filter by layout UUIDs.
</ParamField>

<ParamField query="slugs" type="string[]" required={false}>
    Filter by layout slugs.
</ParamField>

<ParamField query="modes" type="enum[]" required={false}>
    Filter by layout mode: `Form` (form submission), `Signup` (user fields at registration),
    `Custom`, `Welcome`, `Participant` (product participant questionnaire).
</ParamField>

<ParamField query="statuses" type="enum[]" required={false}>
    Filter by layout status: `Draft`, `Published`.
</ParamField>

<ParamField query="productIds" type="integer[]" required={false}>
    Filter by product IDs (for layouts linked to a product, such as participant questionnaires).
</ParamField>

<ParamField query="search" type="string" required={false}>
    Search by layout name. Maximum 50 characters.
</ParamField>

### Sorting

<ParamField query="id" type="enum" required={false}>
    Sort by ID: `ASC` or `DESC`.
</ParamField>

<ParamField query="createdAt" type="enum" required={false}>
    Sort by creation date: `ASC` or `DESC`.
</ParamField>
