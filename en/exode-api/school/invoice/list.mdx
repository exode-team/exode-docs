<HeaderParamFields/>

```
GET /saas/v2/invoice/list/raw
```

<Info>
    All filter parameters are optional. Array parameters are passed by repeating the parameter in the query string:
    `userIds=1&userIds=2`. Ranges (`*Range`) are passed as nested parameters, for example
    `createdAtDateRange[from]` and `createdAtDateRange[to]`.
</Info>

## Request parameters

### Pagination

<ParamField query="skip" type="integer" required={false}>
    Number of records to skip. Defaults to `0`.
</ParamField>

<ParamField query="page" type="integer" required={false}>
    Page number (an alternative to `skip`). Starts at `1`.
</ParamField>

<ParamField query="take" type="integer" required={false}>
    Number of records per page. From `1` to `1000`. Defaults to `100`.
</ParamField>

### Filtering

<ParamField query="invoiceIds" type="integer[]" required={false}>
    Filter by invoice IDs.
</ParamField>

<ParamField query="userIds" type="integer[]" required={false}>
    Filter by user IDs.
</ParamField>

<ParamField query="productIds" type="integer[]" required={false}>
    Filter by product IDs.
</ParamField>

<ParamField query="types" type="enum[]" required={false}>
    Filter by invoice type:
    - `Regular`: one-time payment;
    - `InstallmentInit` / `SubscriptionInit`: setting up an installment plan / subscription (first payment and card binding);
    - `InstallmentPay` / `SubscriptionPay`: recurring automatic charges for an installment plan / subscription.
</ParamField>

<ParamField query="search" type="string" required={false}>
    Search (by user/identifiers). Up to 50 characters; leading and trailing spaces are trimmed.
</ParamField>

<ParamField query="createdAtDateRange" type="object" required={false}>
    Invoice creation date range.

    <Expandable title="createdAtDateRange properties">
        <ParamField query="from" type="string">Start of the range (ISO 8601). Must be ≤ `to`.</ParamField>
        <ParamField query="to" type="string">End of the range (ISO 8601). Must be ≥ `from`.</ParamField>
    </Expandable>
</ParamField>

<ParamField query="totalAmountRange" type="object" required={false}>
    Range of the invoice total amount. At least one of the fields must be set.

    <Expandable title="totalAmountRange properties">
        <ParamField query="from" type="number">Minimum amount.</ParamField>
        <ParamField query="to" type="number">Maximum amount.</ParamField>
    </Expandable>
</ParamField>

<ParamField query="utmParams" type="object" required={false}>
    Filter by the invoice's UTM tags.

    <Expandable title="utmParams properties">
        <ParamField query="value" type="object[]">
            Array of key-value pairs. Each item: `{ "key": "utm_source", "value": "google" }`
            (`key` and `value` are non-empty strings).
        </ParamField>
    </Expandable>
</ParamField>

<ParamField query="payment" type="object" required={false}>
    Filter by the related payment.

    <Expandable title="payment properties">
        <ParamField query="paymentIds" type="integer[]">Payment IDs.</ParamField>
        <ParamField query="acquiringIds" type="integer[]">Acquiring IDs.</ParamField>
        <ParamField query="statuses" type="enum[]">
            Payment statuses: `Created`, `WaitingPay`, `WaitingForBinding`, `Processing`, `Completed`,
            `BindingCompleted`, `Canceled`. An invoice is included if **at least one** of its payments
            (payment attempts) has the specified status.
        </ParamField>
        <ParamField query="actualStatuses" type="enum[]">
            Status of the invoice's **latest** payment; same values as `statuses`. Use it to
            get the current payment state: for example, `payment[actualStatuses]=Completed` returns paid invoices.
        </ParamField>
    </Expandable>
</ParamField>

## Response fields

<ResponseField name="payload" type="object">
    Paginated list of invoices (compact projection). For the full structure, see `invoice` in the
    [`payment`](/en/exode-api/objects/entities/payment) reference.

    <Expandable title="payload properties">
        <ResponseField name="items" type="object[]">
            Array of invoices.

            <Expandable title="Item properties">
                <ResponseField name="invoiceId" type="integer">Invoice ID.</ResponseField>
                <ResponseField name="invoiceUuid" type="string">Invoice UUID.</ResponseField>
                <ResponseField name="type" type="enum">Invoice type (see the `types` filter).</ResponseField>
                <ResponseField name="status" type="enum">
                    Invoice status: `Active` or `Canceled`. This is the status of the invoice itself, not of the payment: `Active` does not
                    mean the invoice is paid. Check payment with the `payment[actualStatuses]` filter or the
                    `PaymentCompleted` webhook.
                </ResponseField>
                <ResponseField name="totalAmount" type="number">Invoice total amount.</ResponseField>
                <ResponseField name="discountAmount" type="number">Discount amount.</ResponseField>
                <ResponseField name="currency" type="enum">Currency: `Free`, `Exes`, `Rub`, `Uzs`, `Kzt`, `Usd`, `Eur`.</ResponseField>
                <ResponseField name="createdAt" type="string">Creation date (ISO 8601).</ResponseField>
                <ResponseField name="expireAt" type="string | null">Expiration date (ISO 8601).</ResponseField>
                <ResponseField name="user" type="object">
                    Buyer: `id`, `tgId`, `login`, `email`, `phone`, `fullName`.
                </ResponseField>
                <ResponseField name="products" type="object[]">
                    Invoice line items: an array of `{ productId, courseId, totalPrice, discountAmount }` objects.
                </ResponseField>
            </Expandable>
        </ResponseField>

        <ResponseField name="page" type="integer">Current page.</ResponseField>
        <ResponseField name="count" type="integer">Total number of records.</ResponseField>
        <ResponseField name="pages" type="integer">Total number of pages.</ResponseField>
        <ResponseField name="isFirst" type="boolean">Whether this is the first page.</ResponseField>
        <ResponseField name="isLast" type="boolean">Whether this is the last page.</ResponseField>
        <ResponseField name="next" type="object">Parameters of the next page (`skip`, `take`, `page`).</ResponseField>
        <ResponseField name="prev" type="object">Parameters of the previous page (`skip`, `take`, `page`).</ResponseField>
    </Expandable>
</ResponseField>
